----- INCLUINDO CONSTRAINTS NAS TABELAS DO CONJUNTO DE DADOS BASE -----

-- TABELA CATEGORIAS:

ALTER TABLE Base.categorias
ADD PRIMARY KEY (ID_Categoria) NOT ENFORCED;


-- TABELA CLIENTES:

ALTER TABLE Base.clientes
ADD PRIMARY KEY (ID_Cliente) NOT ENFORCED;


-- TABELA LOCAIS:

ALTER TABLE Base.locais
ADD PRIMARY KEY (Cidade) NOT ENFORCED;


-- TABELA LOJAS:

ALTER TABLE Base.lojas
ADD PRIMARY KEY (ID_Loja) NOT ENFORCED;
ADD FOREIGN KEY(Loja) references Base.locais(Cidade) NOT ENFORCED;


-- TABELA PRODUTOS:

ALTER TABLE Base.produtos
ADD PRIMARY KEY (ID_Produto) NOT ENFORCED;
ADD FOREIGN KEY(ID_Categoria) references Base.categorias(ID_Categoria) NOT ENFORCED;


-- TABELA PEDIDOS:

ALTER TABLE Base.pedidos
ADD PRIMARY KEY (ID_Pedido) NOT ENFORCED;
ADD FOREIGN KEY(ID_Loja) references Base.lojas(ID_Loja) NOT ENFORCED;
ADD FOREIGN KEY(ID_Produto) references Base.produtos(ID_Produto) NOT ENFORCED;
ADD FOREIGN KEY(ID_Cliente) references Base.clientes(ID_Cliente) NOT ENFORCED;
